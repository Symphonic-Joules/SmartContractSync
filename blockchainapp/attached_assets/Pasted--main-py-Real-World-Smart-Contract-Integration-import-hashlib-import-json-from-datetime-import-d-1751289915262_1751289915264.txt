class RealBlockchainInteractions:

    def _send_transaction(self, contract_function, params: list, gas_limit: int) -> Dict:
        if not self.service_account:
            return {"status": "error", "reason": "Service account not configured (private key missing).", "action": contract_function.__name__}

        try:
            nonce = self.w3.eth.get_transaction_count(self.service_account.address)
            gas_price = self.w3.eth.gas_price

            tx_dict = contract_function(*params).build_transaction({
                'from': self.service_account.address,
                'nonce': nonce,
                'gas': gas_limit,
                'gasPrice': gas_price
            })

            signed_tx = self.w3.eth.account.sign_transaction(tx_dict, private_key=SERVICE_PRIVATE_KEY)
            tx_hash = self.w3.eth.send_raw_transaction(signed_tx.rawTransaction)
            
            print(f"  [Web3] Tx sent: {tx_hash.hex()} for {contract_function.__name__}")
            receipt = self.w3.eth.wait_for_transaction_receipt(tx_hash)
            print(f"  [Web3] Tx mined: {receipt.transactionHash.hex()} status: {receipt.status}")

            if receipt.status == 1:
                return {
                    "transaction_hash": receipt.transactionHash.hex(),
                    "status": "success",
                    "gas_used": receipt.gasUsed,
                    "action": contract_function.__name__,
                    "contract_address": receipt.to.hex() if receipt.to else '0x0'
                }
            else:
                return {
                    "transaction_hash": receipt.transactionHash.hex(),
                    "status": "failed",
                    "reason": "Transaction failed on-chain",
                    "gas_used": receipt.gasUsed,
                    "action": contract_function.__name__
                }
        except Exception as e:
            print(f"  [Web3 ERROR] Failed to execute {contract_function.__name__}: {e}")
            return {"status": "error", "action": contract_function.__name__, "error": str(e)}

# --- Verification Logic (now calls real blockchain interactions) ---
def verify_access_and_trigger_hooks(sigil: GrainSigilMetadata, lock_id: str) -> tuple[bool, str, list]:
    lock = DIGITAL_LOCKS.get(lock_id)
    if not lock:
        return False, f"Error: Lock '{lock_id}' not found.", []

    # Check for service account configuration before attempting any real transactions
    if not real_blockchain_interactions.service_account and lock.on_chain_actions:
        return False, "Denied: Service account (private key) not configured for on-chain actions.", []
    if not w3.is_connected():
        return False, "Denied: Not connected to blockchain network.", []

    # --- Re-adding comprehensive policy checks ---
    sigil_ethics = set(sigil.ethical_profile)
    sigil_capabilities = set(sigil.capabilities)
    sigil_attributes = sigil.user_attributes
    sigil_proofs = sigil.proofs
    sigil_verifiable_credentials = sigil.verifiable_credentials

    access_policy = lock.access_policy

    # Check 1: Ethical Alignment
    required_ethics = set(access_policy.required_ethics)
    if not required_ethics.issubset(sigil_ethics):
        missing_ethics = required_ethics - sigil_ethics
        return False, f"Denied: Missing required ethical declarations: {', '.join(missing_ethics)}", []

    # Check 2: Capability Matching
    required_capabilities = set(access_policy.required_capabilities)
    if not required_capabilities.issubset(sigil_capabilities):
        missing_capabilities = required_capabilities - sigil_capabilities
        return False, f"Denied: Missing required capabilities: {', '.join(missing_capabilities)}", []

    # Check 3: Attribute Conditions
    attribute_conditions = access_policy.attribute_conditions
    for attr, required_value in attribute_conditions.items():
        if attr == "sigil_fingerprint_present":
            if required_value is True and not sigil.sigil_fingerprint:
                return False, f"Denied: Lock requires 'sigil_fingerprint_present' but it is missing.", []
            if required_value is False and sigil.sigil_fingerprint:
                 return False, f"Denied: Lock requires 'sigil_fingerprint_present' to be false, but it is present.", []
            continue
        
        if attr not in sigil_attributes:
            return False, f"Denied: Missing required attribute '{attr}'", []
        
        sigil_attr_value = sigil_attributes[attr]
        
        if attr.endswith("_min"):
            base_attr = attr[:-4]
            if base_attr not in sigil_attributes or sigil_attributes[base_attr] < required_value:
                return False, f"Denied: Attribute '{base_attr}' value ({sigil_attributes.get(base_attr)}) is below required minimum ({required_value})", []
        elif attr.endswith("_max"):
             base_attr = attr[:-4]
             if base_attr not in sigil_attributes or sigil_attributes[base_attr] > required_value:
                return False, f"Denied: Attribute '{base_attr}' value ({sigil_attributes.get(base_attr)}) is above required maximum ({required_value})", []
        elif sigil_attr_value != required_value:
            return False, f"Denied: Attribute '{attr}' value '{sigil_attr_value}' does not match required '{required_value}'", []

    # Check 4: Proof Verification
    requires_proof_type = access_policy.requires_proof_type
    if requires_proof_type:
        if not sigil_proofs.get(requires_proof_type):
            return False, f"Denied: Missing required proof type '{requires_proof_type}'", []

    # Check 5: Verifiable Credentials
    required_credentials = access_policy.required_credentials
    for req_cred in required_credentials:
        cred_type = req_cred.get("type")
        cred_found = False
        for sigil_cred in sigil_verifiable_credentials:
            if sigil_cred.get("type") == cred_type:
                cred_found = True
                break
        if not cred_found:
            return False, f"Denied: Missing required verifiable credential of type '{cred_type}'", []

    # --- All access checks pass, trigger real on-chain actions ---
    receipts = []
    for action in lock.on_chain_actions:
        func_name = action.function
        params = []
        
        # Resolve parameters dynamically
        for p in action.params:
            if p == "linked_wallet_address":
                params.append(sigil.linked_wallet_address)
            elif p == "testimony_proof_hash_val": # For example, passing the proof hash as a data param
                params.append(sigil.proofs.get("testimony_proof", "0x0"))
            elif p == "sigil_identity_hash_val":
                params.append(sigil.identity_proof_hash)
            elif p == "user_role_val": # Example: pass user_role from attributes
                params.append(sigil.user_attributes.get("user_role", "default_role"))
            else:
                params.append(p) # Assume literal value

        try:
            # Call the real blockchain interaction method
            # IMPORTANT: The function names here (e.g., 'allocate_reparations_real') MUST match
            # the methods in RealBlockchainInteractions class.
            action_func = getattr(real_blockchain_interactions, func_name)
            receipt = action_func(*params)
            receipts.append(receipt)
            # Update simulated wallet state if real transaction was successful (for frontend display)
            if receipt.get("status") == "success":
                target_wallet = blockchain_simulator_stub._get_wallet(sigil.linked_wallet_address)
                if receipt["action"] == "allocate_reparations_real":
                    # You need to parse the amount from the original params or receipt data if complex
                    amount_allocated = next((p for p in action.params if isinstance(p, (int, float))), 0)
                    target_wallet["CareTokens"] = target_wallet.get("CareTokens", 0.0) + amount_allocated
                elif receipt["action"] == "mint_testimony_credit_real":
                    target_wallet["NFTs"].append(NftAsset(type="TestimonyCredit", id=receipt['transaction_hash'][:8], impact_tag=params[1], token_uri=params[2] if len(params)>2 else None).dict())
                elif receipt["action"] == "mint_creative_work_real":
                     target_wallet["NFTs"].append(NftAsset(type="CreativeWork", id=params[1], token_uri=params[2] if len(params)>2 else None).dict())
                # Add other updates for simulated_wallets based on real actions here

        except Exception as e:
            print(f"  [Real Tx ERROR] Function '{func_name}' execution failed: {e}")
            receipts.append({"status": "error", "action": func_name, "error": str(e)})

    return True, lock.access_message, receipts

# NEW: Proxy endpoint for fetching NFT metadata
@app.get("/nft_metadata_proxy")
async def nft_metadata_proxy(token_uri: str = Query(...)):
    if not token_uri.startswith("ipfs://"):
        raise HTTPException(status_code=400, detail="Only IPFS token URIs are supported.")
    
    ipfs_hash = token_uri.split('ipfs://')[1]
    gateway_url = f"https://cloudflare-ipfs.com/ipfs/{ipfs_hash}"
    
    async with httpx.AsyncClient() as client:
        try:
            response = await client.get(gateway_url, timeout=10.0)
            response.raise_for_status()  # Raise an exception for bad status codes
            return response.json()
        except httpx.RequestError as e:
            raise HTTPException(status_code=502, detail=f"Failed to fetch metadata from IPFS gateway: {e}")
        except json.JSONDecodeError:
            raise HTTPException(status_code=500, detail="Failed to parse metadata JSON.")
